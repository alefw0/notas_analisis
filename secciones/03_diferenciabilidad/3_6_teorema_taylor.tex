\section{Teorema de Taylor}

Como hemos observado anteriormente, si ${f}$ es diferenciable en ${c}$, entonces ${f}$ es aproximadamente una función real a las proximidades de ${c}$. Esto es, la ecuacion:\
 
\begin{center}
    ${f(x)=f(c)+f'(c)(x-c)}$ \
\end{center}
es aproximadamente correcto cuando ${(x-c)}$ es pequeño. Pero, que pasa cuando conocemos la segunda derivada o la tercer derivada e incluso que pasa si ${f}$ tiene n-ésima derivada.  \\\\
Pues bien el teorema de Taylor nos dice que, en general, ${f}$ puede aproximarse por medio de un polinomio de grado ${n-1}$, si ${f}$ posee derivadas hasta de orden n. Además, el teorema de Taylor proprciona una expresión util para calcular el error cometido en esta aproximación. \\\\
El teorema de Taylor se obtiene como consecuencia de un resultado más general, veamos entonces el siguiente teorema\\\\\\
\begin{teorema}
    Sean ${f}$ y ${g}$ dos funciones que poseen derivadas n-ésimas ${f^{(n)}}$ y ${g^{(n)}}$ finitas en un intervalo abierto ${(a,b)}$ y derivadas ${(n-1)}$ continuas en el intervalo cerrado ${[a,b]}$, supongamos que ${c\epsilon[a,b]}$. Entonces, para todo ${x}$ de ${[a,b]}$, ${x\neq c}$, existe un punto ${x_{1}}$ interior al intervalo que une a ${x}$ con ${c}$ tal que:\\


\begin{center}
    ${\left[ f(x)-\sum_{k=0}^{n-1}\frac{f^{(n)}(c)}{k!}(x-c)^{k} \right]g^{(n)}(x_{1})=f^{(n)}(x_{1})\left[  g(x)-\sum_{k=0}^{n-1}\frac{g^{(n)}(c)}{k!}(x-c)^{k}\right]}$ \
\end{center}
\end{teorema}
\begin{prueba}
    Supongamos que ${c\in [a,b]}$ y tomamos ${{x}\geq {c}}$ con ${x}$ fijo.
Definamos ahora las funciones auxiliares ${F(t)}$ y ${G(t)}$ para cada ${t\epsilon[c,x]}$ como:

\begin{center}
    ${F(t)= f(t)+\sum_{k=1}^{n-1}\frac{f^{(k)}(t)}{k!}(x-t)^{k}}$ \
\end{center}
\begin{center}
    ${G(t)= g(t)+\sum_{k=1}^{n-1}\frac{g^{(k)}(t)}{k!}(x-t)^{k}}$ 
\end{center}\
\textbf{*} Cuando ${t=x}$, la suma se anula porque ${(x-x)=0}$, entonces obtenemos ${F(x)=f(x)}$ y ${G(x)=g(x)}$\\\\\
\textbf{*} Cuando ${t=c}$, tenemos\
\begin{center}
    ${G(c)= g(c)+\sum_{k=1}^{n-1}\frac{g^{(k)}(c)}{k!}(x-c)^{k}}$ 
\end{center}\
\begin{center}
    ${G(c)= g(c)+g'(c)(x-c)+\frac{g^{(2)}(c)}{2!}(x-c)^{2}+\frac{g^{(3)}(c)}{3!}(x-c)^{3}+...+\frac{g^{(n-1)}(c)}{(n-1)!}(x-c)^{n-1}}$ 
\end{center}\
\begin{center}
    ${G(c)= \sum_{k=0}^{n-1}\frac{g^{(k)}(c)}{k!}(x-c)^{k}}$ 
\end{center}\
y asi, mismo:
\begin{center}
    ${F(c)= \sum_{k=0}^{n-1}\frac{f^{(k)}(c)}{k!}(x-c)^{k}}$ 
\end{center}\
Que son precisamente los polinomios de Taylor de grado ${n-1}$\\\\
Aplicamos el Teorema del valor medio generalizado que nos indica que existe ${x_{1}\in(c,x)}$ tal que:
\begin{center}
    ${F'(x_{1})\left[ G(x)-G(c) \right]=G'(x_{1})\left[ F(x)-F(c) \right]}$ (*)
\end{center}\
\\Sustituimos ${F(x)=f(x)}$ y ${G(x)=g(x)}$

\begin{center}
    ${F'(x_{1})\left[ g(x)-G(c) \right]=G'(x_{1})\left[ f(x)-F(c) \right]}$
\end{center}\
Usando la sustitución ${t=x_{1}}$, calculamos ${F'(t)}$ y ${G'(t)}$\
Derivamos ${F(t)}$ (para ${G(t)}$ es análogo) para un termino con ${k}$ fijo.
\begin{center}
     ${\frac{d }{dx}\left[ \frac{f^{(k)}(t)}{k!}(x-t)^{k} \right]=\frac{f^{(k+1)}(t)}{k!}(x-t)^{k}+\frac{f^{(k)}(t)}{k!}k(x-t)^{k-1}(-1)}$
\end{center}\
\begin{center}
     ${=\frac{f^{(k+1)}(t)}{k!}(x-t)^{k}-\frac{f^{(k)}(t)}{(k-1)!}(x-t)^{k-1}}$
\end{center}\
Ahora, sumamos para ${k=1}$ hasta ${n-1}$

\begin{center}
    ${F'(t)=f'(t)+\sum_{k=1}^{n-1}\frac{f^{(k+1)}(t)}{k!}(x-t)^{k}-\sum_{k=1}^{n-1}\frac{f^{(k)}(t)}{(k-1)!}(x-t)^{k-1}}$
\end{center}\
Hacemos un cambio de indice para la segunda sumatoria, tomamos ${j=k-1}$, con ${j=0}$ hasta ${n-1}$

\begin{center}
    ${\sum_{k=1}^{n-1}\frac{f^{(k)}(t)}{(k-1)!}(x-t)^{k-1}=\sum_{j=0}^{n-2}\frac{f^{(j+1)}(t)}{j!}(x-t)^{j}}$ \
\end{center}\
y asi, tenemos 
\begin{center}
    ${F'(t)=f'(t)+\sum_{k=1}^{n-1}\frac{f^{(k+1)}(t)}{k!}(x-t)^{k}-\sum_{j=0}^{n-2}\frac{f^{(j+1)}(t)}{j!}(x-t)^{j}}$ \
\end{center}\
Observemos que el termino ${f'(t)}$ corresponde a ${j=0}$ en esta suma. Por lo tant, al restar, todos los terminos desde ${k=0}$ hasta ${n-2}$ se cancelan,subreviviendo unicamente el ultimo termino de la primera suma, cuando ${k=n-1}$, obteniendo asi:
\begin{center}
    ${F'(t)=\frac{f^{(n)}(t)}{(n-1)!}(x-t)^{n-1}}$ \
\end{center}\
Análogamente, tenemos
\begin{center}
    ${G'(t)=\frac{g^{(n)}(t)}{(n-1)!}(x-t)^{n-1}}$ \
\end{center}\
Recordando que ${t=x_{1}}$, y sustituyendo ${F'(x)}$ y ${G'(x)}$ en (*), tenemos
\begin{center}
    ${\frac{f^{(n)}(x_{1})}{(n-1)!}(x-1)^{n-1}\left[ g(x)-G(c) \right]=\frac{g^{(n)}(x_{1})}{(n-1)!}(x-t)^{n-1}\left[ f(x)-F(c) \right]}$
\end{center}
\begin{center}
    ${f^{(n)}(x_{1})\left[ g(x)-G(c) \right]=g^{(n)}(x_{1})\left[ f(x)-F(c) \right]}$
\end{center}
Reemplazando ${F(c)}$ y ${G(c)}$, obtenemos 
\begin{center}
    ${f^{(n)}(x_{1})\left[ g(x)-\sum_{k=0}^{n-1}\frac{g^{(k)}(c)}{k!}(x-c)^{k} \right]=g^{(n)}(x_{1})\left[ f(x)-\sum_{k=0}^{n-1}\frac{f^{(k)}(c)}{k!}(x-c)^{k} \right]}$
\end{center}
 y esto, es equivalente a tener
 \begin{center}
    ${\left[ f(x)-\sum_{k=0}^{n-1}\frac{f^{(k)}(c)}{k!}(x-c)^{k} \right]g^{(n)}(x_{1})=f^{(n)}(x_{1})\left[ g(x)-\sum_{k=0}^{n-1}\frac{g^{(k)}(c)}{k!}(x-c)^{k} \right]}$
\end{center}
\end{prueba}
\begin{teorema}
     Sea ${f}$ una función que admita derivada n-ésima ${f^{(n)}}$ finita en todo el intervalo abierto ${(a,b)}$ y supongamos que $f^{(n-1)}$ es continua en el intervalo cerrado ${[a,b]}$. Supongamos que ${c\epsilon[a,b]}$. Entonces, para todo ${x\epsilon[a,b]}$, ${x\neq0}$, existe un punto ${x_{1}}$, interior al intervalo que une ${x}$ con ${c}$ tal que:\\
 \begin{center}
${f(x)=f(c)+\sum_{k=1}^{n-1}\frac{f^{(k)}(c)}{k!}(x-c)^{k}+\frac{f^{(n)}(x_{1})}{n!}(x-n)^{n}}.$
\end{center}
\end{teorema}
\begin{prueba}
    Este teorema es un caso particular del teorema anterior, si consideramos ${g(x)=(x-c)^{n}}$
Veamos lo siguiente. Dadas ${f}$ y ${g}$ funciones con derivadas n-ésimas finitas en ${(a,b)}$ y derivadas ${(n-1)}$-ésimas continuas en ${[a,b]}$ para todo ${x\epsilon[a,b]}$, ${x\neq c}$, existe ${x_{1}}$ en el intervalo abierto que une ${x}$ y ${c}$ tal que:
\begin{equation}\label{taylor}
    {\left[ f(x)-\sum_{k=0}^{n-1}\frac{f^{(k)}(c)}{k!}(x-c)^{k} \right]g^{(n)}(x_{1})=f^{(n)}(x_{1})\left[ g(x)-\sum_{k=0}^{n-1}\frac{g^{(k)}(c)}{k!}(x-c)^{k} \right]}
\end{equation}

De la expresión
\begin{center}
    ${f^{(n)}(x_{1})\left[ g(x)-\sum_{k=0}^{n-1}\frac{g^{(k)}(c)}{k!}(x-c)^{k} \right]}$  
\end{center}
Obtenemos
\begin{equation}\label{tay1}
{f^{(n)}(x_{1})(x-c)^{n}}
\end{equation}
Ya que de ${g(x)=(x-c)^{n}}$, se cumple que:\\
\begin{center}${g(c)=0}$\\
${g'(c)=0}$ ... ${g^{(k)}(c)=0}$\
\end{center}
Por lo tanto\\\\
\begin{center}
${\sum_{k=0}^{n-1}\frac{g^{(k)}(c)}{k!}(x-c)^{k}=0}$\
\end{center}
Ahora bien para ${g(x)=(x-c)^{n}}$\\
Calculamos su derivada n-ésima\\
\begin{center}
${g'(x)=n(x-c)^{n-1}}$\\
${g^{(2)}(x)=n(n-1)(x-c)^{n-2}}$\\
.\\
.\\
.\\
${ g^{(k)}(x)=n(n-1)(n-2)...(x-k+1)(x-c)^{n-k}}$\\
\end{center}
Cuando ${k=n}$, ${(x-c)^{0}=1}$, quedando asi el producto
\begin{center}
    ${n(n-1)(n-2)\dots2 \cdot 1=n! }$
\end{center}
Por lo tanto para cualquier punto ${x_{1}}$
\begin{equation}\label{tay2}
    g^{(n)}(x_{1})=n!  
\end{equation}
Sustituyendo \eqref{tay1} y \eqref{tay2} en \eqref{taylor} se tiene lo siguiente
\begin{center}
    ${\left[ f(x)-\sum_{k=0}^{n-1}\frac{f^{(k)}(c)}{k!}(x-c)^{k} \right]n!=f^{(n)}(x_{1})(x-c)^{n}},$
\end{center}
luego 
\begin{center}
    ${f(x)-\sum_{k=0}^{n-1}\frac{f^{(k)}(c)}{k!}(x-c)^{k} =\frac{f^{(n)}(x_{1})}{n!}(x-c)^{n}}.$
\end{center}
Asi, 
\begin{center}
    ${f(x)=\sum_{k=0}^{n-1}\frac{f^{(k)}(c)}{k!}(x-c)^{k}+\frac{f^{(n)}(x_{1})}{n!}(x-c)^{n}}$
\end{center}
que es lo queremos probar.
\end{prueba}
